
/* === Rendering baseline (works across sites) === */
/* These rules normalize spacing, sizing, scrollbars, focus rings and form
   controls so the same page looks steadier on dark and light sites, on
   narrow handheld windows and on desktop preview. They are scoped to the
   parts of the builder and preview that the editor controls, so they do
   not leak into unrelated third-party widgets that sites may embed here. */

  text-rendering: optimizeLegibility;
  -webkit-font-smoothing: antialiased;
  -moz-osx-font-smoothing: grayscale;
}

  text-size-adjust: 100%;
  -webkit-text-size-adjust: 100%;
  line-height: inherit;
  margin: 0;
}

  font: inherit;
}

  display: block;
  max-width: 100%;
  height: auto;
}

  object-fit: contain;
}

/* Builder element cards on the canvas, not in the sidebar. These read as
   selected tiles while editing and loosen up a little when the preview hides
   the inspector chrome.

   Field wrapper boundary
   The outer wrapper for a styled text field originally stopped the builder
   at `border-radius: 4px`. Text with descenders like `g`, `y`, or `p` then
   clipped under the corners and looked cut even though the characters were
   fully there. Round the wrapper the same way the inner input is rounded so
   the visible edge matches the editable area. */

#canvas .element.is-input,
#canvas .element.is-input *,
#canvas .element.is-input::before,
#canvas .element.is-input::after {
  --element-radius: var(--radius-md);
}

#canvas .element.is-input {
  border-radius: var(--element-radius);
}

#canvas .element.is-input .element-value,

  border-radius: inherit;
}
